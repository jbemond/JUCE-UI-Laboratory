cmake_minimum_required(VERSION 3.22)
project(JUCEUILaboratory VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.9 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_gui_app(JUCEUILaboratory PRODUCT_NAME "JUCE UI Laboratory" COMPANY_NAME "JB Emond" BUNDLE_ID "fr.jbemond.juceuilaboratory")
juce_generate_juce_header(JUCEUILaboratory)
target_sources(JUCEUILaboratory PRIVATE src/Main.cpp src/MainComponent.cpp src/MainComponent.h)
target_compile_definitions(JUCEUILaboratory PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(JUCEUILaboratory PRIVATE juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
